// =================== CORE & DEBUGGING ===================

insbot set debug                     "0" // print bot debug info to console
insbot set draw                      "0" // through-wall bot debug overlays
insbot set draw_path                 "0" // also draw nav paths (green)
insbot set draw_team                 "1" // reference team for objective colors: green=friendly, grey=neutral, red=hostile
insbot set freeze                    "0" // freeze all bot movement and actions (1=frozen, overridden by force pathfinding)
insbot set profile                   "0" // built-in per-frame profiler: measures BuildPath/GetPathCost/FindArea, traces, hearing, teamsight and prints a summary every 5 s (0=off)
insbot set quota                     "0" // auto-maintain this many bots (0=manual)
insbot set reserve                   "2" // keep this many free slots open for human players

// =================== SPAWNING & SETUP ===================

insbot set dead_cmds                 "creinforce;declinepromo" // commands sent every 5s while dead

insbot set join_cmds_b               "pteamsetup 2;pfullsetup {f} {r};creinforce" // side-B join/spawn commands

insbot set join_interval             "1.0" // seconds between join commands (minimum 0.05)
insbot set join_timeout              "40" // kick+retry a bot that can't get into a squad after this many seconds (0=never)
insbot set quota_add_interval        "0.25" // seconds between quota replacement bot creations after startup
insbot set quota_start_interval      "0.20" // seconds between initial quota client creations; joins release together after the roster is complete (minimum 0.05)
insbot set slot_sweep                "20" // how many squad slots {s} cycles through when joining
insbot set spawn_nocollide           "1" // temporarily make newly spawned bots nonblocking to other grace bots (0=off)
insbot set spawn_nocollide_time      "5.0" // unfrozen simulation seconds before spawn collision is restored once the hull is clear
insbot set randomize_customization    "1" // randomize faction-valid player skins/accessories on each spawn (0=off)
insbot set randomize_loadout          "1" // randomize faction- and role-valid primary weapon loadouts on each spawn (0=off; sends -1)

insbot set doorway_teleport          "1" // warp a stalled front bot past a detected doorway after doorway_teleport_time (0=off)
insbot set doorway_teleport_time     "3.0" // continuous doorway no-progress seconds before a full-hull-validated teleport
insbot set ladder_lip_tol           "4.0" // vertical tolerance (units) for latching the ladder dismount near the recorded lip; clamped to 0..6, above 6 the bot latches mid-shaft and thrashes on the 18u lost-height guard

insbot set ladder_queue              "1" // single-owner queues at both ends of typed NAV ladders (0=off)
insbot set ladder_up_penalty         "500" // extra A* cost for climbing UP a ladder; descending is the easy direction
insbot set move_speed                "400" // movement input magnitude
insbot set nav                       "1" // enable/disable nav mesh pathfinding system (0=turn off nav system, overridden by force pathfinding)
insbot set sprint                    "1" // hold sprint (IN_ALT2) while moving with no enemy contact; game stamina rules apply
insbot set streaming                 "2" // corridor queues (0=off; 1=all narrow NAV runs; 2=automatic at full-hull-verified physical doorways/corridors)
insbot set stuck_dist                "16" // inches of progress toward the next waypoint that counts as moving
insbot set stuck_nocollide           "1" // try local player-body collision grace before stuck or doorway teleport (0=off)
insbot set stuck_time                "1.0" // no-progress seconds before body-grace or broken-link recovery starts

// =================== HUMVEE AUTO-TURRET ===================
// The Humvee's M249 is unmanned and automatic. A Humvee is spawned inert; the
// turret arms the first time a player on a play team (1 or 2) sits in the
// driver seat, and that player's team owns it from then on. Leaving the vehicle
// does not disarm it, and boarding an enemy turret does not steal it.
//
// These settings are read live. A per-turret override can be set with
//   insbot turret set <entindex> <name> <value>
// and "all" writes the global instead:
//   insbot turret set all <name> <value>
//
// turret_damage is per BULLET and the turret is fully automatic - while it
// holds a target it fires continuously at the M249's 800 RPM, with no burst or
// pause cycle - so sustained damage is far higher than a single hit. The real
// M249 round in this mod does about 151 damage; the default of 50 is a
// deliberate playtest value, not a simulation of the weapon.
// See docs/humvee-turret-damage-audit.md.

insbot set turret_enabled              "1"     // master switch for the Humvee auto-turret (0=off)
insbot set turret_aim_speed            "90"    // turret traverse speed (deg/s)
insbot set turret_aim_error            "4.0"   // turret aim error in degrees (0=perfect)
insbot set turret_engage_range         "1500"  // turret firing gate distance (units)
insbot set turret_damage               "50"    // turret damage per bullet
insbot set turret_scan_speed           "20"    // turret idle scan speed (deg/s)
insbot set turret_acquire_fov          "360"   // turret acquisition cone (deg; 360=omnidirectional)
insbot set turret_fire_tolerance_yaw    "5.0"  // turret fire gate: max yaw error (deg)
insbot set turret_fire_tolerance_pitch  "4.0"  // turret fire gate: max pitch error (deg)
insbot set turret_spread               "0.230" // turret bullet spread cone (deg; M249 real value)

// =================== BOT ANTI-VEHICLE POLICY ===================

// A Humvee is owned by the team of the FIRST player to board it, and it stays
// on that team for the rest of its life - including after the driver dies, and
// even if an enemy later climbs in. Bots engage a truck owned by the other
// team, except while a teammate is sitting inside it.
//
// A seated driver is immune to bullet fire (the engine excludes
// COLLISION_GROUP_IN_VEHICLE from traces), so the truck's 400 HP hull is the
// only thing that can actually be damaged. A destroyed wreck is dropped
// immediately, and an unoccupied or never-boarded truck is never engaged.
//
// At sm_humvee_armor_scale 0.05 a rifle round does about 1.75 damage to the
// hull - roughly 230 hits to break one truck. Only the RPG-7 and M203 bypass
// that scale. Set 1 to restrict bots to those weapons; the default 2 lets any
// weapon engage, but only when no infantry threat is visible.

insbot set target_vehicles            "2"     // 0=ignore Humvees, 1=anti-armor weapons only (RPG/M203), 2=all weapons when no infantry threat is visible (default)

// =================== ENGINE OFFSETS (ADVANCED) ===================

// DO NOT COPY THIS SECTION INTO YOUR SERVER CONFIG, AND DO NOT CHANGE THESE
// VALUES UNLESS YOU KNOW WHAT YOU ARE DOING.
//
// These are raw byte offsets into the game's own C++ objects. The defaults are
// the ones verified against the shipped server binaries for this plugin
// version, per platform. A wrong offset does not fail loudly: the plugin reads
// whatever bytes happen to sit there, so objectives can silently report the
// wrong owner, and bots may push to or defend the wrong points. Some of these
// fields are read through a memory-safety guard, so a bad value usually
// degrades to "unreadable" rather than crashing - but that still breaks the
// feature silently.
//
// Only touch these if you are re-targeting the plugin at a different game
// build and have re-derived the offsets from that build's server binary.
// Offsets are NOT interchangeable between Windows and Linux.
